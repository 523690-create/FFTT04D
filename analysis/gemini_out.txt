### (1) Collapsible/Expandable/Pop-out Recordings Panel

Use a `JSplitPane` for collapsibility and a dedicated `JPanel` as a container for `recordingsList` to facilitate reparenting.

```kotlin
// In AnalyzerWindow class
private val recordingsPanelContainer = JPanel(BorderLayout()) // Container for recordingsList
private val recordingsList = JList<String>() // Your existing JList
private val mainSplitPane = JSplitPane(JSplitPane.HORIZONTAL_SPLIT)

init {
    recordingsPanelContainer.add(JScrollPane(recordingsList), BorderLayout.CENTER)
    mainSplitPane.leftComponent = recordingsPanelContainer
    // mainSplitPane.rightComponent = ... (your main content)
    // add(mainSplitPane, BorderLayout.CENTER)

    // Add a button/menu item to toggle collapse/expand
    val toggleButton = JButton("Toggle Recordings Panel")
    toggleButton.addActionListener {
        mainSplitPane.dividerLocation = if (mainSplitPane.dividerLocation > 0) 0 else 200 // Adjust 200 as needed
    }

    // Add a button/menu item to pop out
    val popOutButton = JButton("Pop Out Recordings")
    popOutButton.addActionListener { popOutRecordingsPanel() }
}

private var detachedFrame: JFrame? = null

private fun popOutRecordingsPanel() {
    if (detachedFrame == null) {
        // 1. Remove from current parent
        mainSplitPane.remove(recordingsPanelContainer)
        mainSplitPane.dividerLocation = 0 // Collapse the space it occupied

        // 2. Create new JFrame
        detachedFrame = JFrame("Recordings").apply {
            size = Dimension(300, 600)
            locationRelativeTo(this@AnalyzerWindow)
            add(recordingsPanelContainer, BorderLayout.CENTER) // Add the same component
            isVisible = true
            // Handle closing: re-dock or just hide
            defaultCloseOperation = WindowConstants.DO_NOTHING_ON_CLOSE
            addWindowListener(object : WindowAdapter() {
                override fun windowClosing(e: WindowEvent?) {
                    dockRecordingsPanel()
                }
            })
        }
    } else {
        detachedFrame?.toFront() // Bring to front if already popped out
    }
}

private fun dockRecordingsPanel() {
    detachedFrame?.apply {
        // 1. Remove from detached frame
        remove(recordingsPanelContainer)
        dispose()
    }
    detachedFrame = null

    // 2. Add back to main window
    mainSplitPane.leftComponent = recordingsPanelContainer
    mainSplitPane.dividerLocation = 200 // Restore default size
    mainSplitPane.revalidate()
    mainSplitPane.repaint()
}
```

```kotlin
// Asynchronous thumbnail loading (using Kotlin Coroutines)
private fun loadThumbnailAsync(recording: AudioRecording, targetWidth: Int, targetHeight: Int, callback: (ImageIcon?) -> Unit) {
    val cached = thumbnailCache[recording.id]
    if (cached != null) {
        callback(cached)
        return
    }

    // Use a background dispatcher for heavy work, then switch to Swing for UI update
    CoroutineScope(Dispatchers.IO).launch {
        try {
            val pcm = AudioDecoder.decode(recording.audioFile)
            if (pcm != null) {
                val tempPngFile = File.createTempFile("spectrogram", ".png")
                SpectrogramRenderer.renderFftPng(pcm, recording.sampleRate, tempPngFile)
                val originalImage = ImageIO.read(tempPngFile)
                tempPngFile.delete() // Clean up temp file

                val scaledImage = originalImage.getScaledInstance(targetWidth, targetHeight, Image.SCALE_SMOOTH)
                val icon = ImageIcon(scaledImage)
                thumbnailCache[recording.id] = icon
                withContext(Dispatchers.Swing) {
                    callback(icon)
                }
            } else {
                withContext(Dispatchers.Swing) { callback(null) }
            }
        } catch (e: Exception) {
            println("Error loading thumbnail for ${recording.id}: ${e.message}")
            withContext(Dispatchers.Swing) { callback(null) }
        }
    }
}
```

### (3) Render MFCC Heatmap Natively

Create a `BufferedImage`, draw on its `Graphics2D`, and display it in a `JLabel`.

```kotlin
import java.awt.*
import java.awt.image.BufferedImage
import javax.swing.*
import kotlin.math.*

// In AnalyzerWindow or a dedicated utility class
fun createMfccHeatmap(mfccData: Array<FloatArray>, width: Int, height: Int): BufferedImage {
    if (mfccData.isEmpty() || mfccData[0].isEmpty()) {
        return BufferedImage(width, height, BufferedImage.TYPE_INT_RGB).apply {
            val g = createGraphics()
            g.color = Color.BLACK
            g.fillRect(0, 0, width, height)
            g.dispose()
        }
    }

    val numFrames = mfccData.size
    val numCoeffs = mfccData[0].size

    // 1. Find min/max for normalization
    var minVal = Float.MAX_VALUE
    var maxVal = Float.MIN_VALUE
    mfccData.forEach { frame ->
        frame.forEach { coeff ->
            minVal = min(minVal, coeff)
            maxVal = max(maxVal, coeff)
        }
    }

    val range = maxVal - minVal
    val midVal = (minVal + maxVal) / 2f // Midpoint for diverging colormap

    val image = BufferedImage(width, height, BufferedImage.TYPE_INT_RGB)
    val g2d = image.createGraphics()

    // Calculate cell dimensions
    val cellWidth = width.toFloat() / numFrames
    val cellHeight = height.toFloat() / numCoeffs

    for (frameIdx in 0 until numFrames) {
        for (coeffIdx in 0 until numCoeffs) {
            val value = mfccData[frameIdx][coeffIdx]

            // 2. Normalize and map to color
            val normalizedValue: Float
            val color: Color

            if (range == 0f) { // All values are the same
                color = Color.GRAY
            } else {
                // Diverging colormap: e.g., Blue (low) -> White (mid) -> Red (high)
                normalizedValue = (value - minVal) / range // 0.0 to 1.0

                // Interpolate between colors
                color = when {
                    value < midVal -> { // Blue to White
                        val t = (value - minVal) / (midVal - minVal) // 0.0 to 1.0 for lower half
                        Color(
                            (255 * t + 0 * (1 - t)).toInt(), // R: 0 -> 255
                            (255 * t + 0 * (1 - t)).toInt(), // G: 0 -> 255
                            (255 * t + 255 * (1 - t)).toInt() // B: 255 -> 255
                        )
                    }
                    else -> { // White to Red
                        val t = (value - midVal) / (maxVal - midVal) // 0.0 to 1.0 for upper half
                        Color(
                            (255 * t + 255 * (1 - t)).toInt(), // R: 255 -> 255
                            (0 * t + 255 * (1 - t)).toInt(),  // G: 255 -> 0
                            (0 * t + 255 * (1 - t)).toInt()   // B: 255 -> 0
                        )
                    }
                }
            }

            g2d.color = color
            g2d.fillRect(
                (frameIdx * cellWidth).toInt(),
                (height - (coeffIdx + 1) * cellHeight).toInt(), // Y-axis inverted for coeffs (0 at bottom)
                ceil(cellWidth).toInt(),
                ceil(cellHeight).toInt()
            )
        }
    }

    g2d.dispose()
    return image
}
```
